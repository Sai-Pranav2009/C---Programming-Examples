PRINTING A HOLLOW RECTANGLE:-
#include<stdio.h>
int main()
{
    int T;
    scanf("%d", &T);
    for(int t = 0; t < T; t++)
        {
            int L, W;
            scanf("%d %d", &L, &W);
            if(t > 0)
                printf("\n");
            for(int i = 1; i<= L; i++)
                {
                    for(int j = 1; j <= W; j++)
                        {
                            if(i == 1 || i == L || j == 1 || j == W)
                                printf("*");
                            else
                                printf(" ");
                        }
                    printf("\n");
                }
        }
    return 0;
}



PRINTING A RIGHT-ANGLED TRIANGLE(WITH NUMBERS):-
#include<stdio.h>
int main()
{
    int T;
    scanf("%d", &T);
    while(T--)
        {
            int N;
            scanf("%d", &N);
            int num = 1;
            for(int i = 1; i <= N; i++)
                {
                    for(int j = 1; j <= i; j++)
                        {
                            printf("%d", num);
                            if(j < i)
                                printf(" ");
                            num++;
                        }
                    printf("\n");
                }
            if(T > 0)
                printf("\n");
        }
    return 0;
}


PRINTING A RIGHT-ANGLED TRIANGLE(WITH '*'):-
#include<stdio.h>
int main()
{
    int T;
    scanf("%d", &T);
    for(int k = 1; k <= T; k++)
        {
            int N;
            scanf("%d", &N);
            printf("Case #%d:\n", k);
            for(int i = 1; i <= N; i++)
                {
                    for(int j = 1; j <= N - i; j++)
                        printf(" ");
                    for(int j = 1; j <= i; j++)
                        printf("*");
                    printf("\n");
                }
        }
    return 0;
}
